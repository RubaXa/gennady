import fs from 'node:fs';
import { createHash } from 'node:crypto';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { buildAuthoringFixture, checkAuthoringBenchmark, getAuthoringCase } from '/private/tmp/gennady-format-oct23.xNC5Zx/plan/ai/flow-eval/authoring-benchmark.ts';
const own=dirname(fileURLToPath(import.meta.url));
const repo='/private/tmp/gennady-format-oct23.xNC5Zx/plan';
const sha=(b:Buffer|string)=>createHash('sha256').update(b).digest('hex');
const bytes=(p:string)=>{const s=fs.lstatSync(p);if(!s.isFile()||s.isSymbolicLink()||s.size>8*1024*1024)throw Error('UNSAFE_REPLAY_INPUT');return fs.readFileSync(p)};

const scratch=fs.mkdtempSync(join(own,'oracle-v6-replay-')), rows:any[]=[];

try {

    const record=json(join(own,'records',`${index}.json`)),result=join(own,'permanent',record.primary.directory);
    const summaryBytes=bytes(join(result,'summary.json'));
    if(sha(summaryBytes)!==record.primary.summarySha256)throw Error('PRIMARY_HASH_DRIFT');

    const artifact=bytes(join(result,item.target));
    for(const mode of ['current-reference-fixture','available-saved-references']) {
      const fixture=fs.mkdtempSync(join(scratch,'fixture-'));created++;
      try {
        const put=(p:string,b:Buffer|string)=>{const target=join(fixture,p);fs.mkdirSync(dirname(target),{recursive:true});fs.writeFileSync(target,b)};
        const saved:any[]=[];
        for(const [path,text] of Object.entries(buildAuthoringFixture(caseId))) {
          const p=join(result,path);
          if(mode==='available-saved-references'&&fs.existsSync(p)){const b=bytes(p);put(path,b);saved.push({path,sha256:sha(b)})}
          else put(path,text);
        }
        put(item.target,artifact);
        const v5=checkAuthoringBenchmark(fixture,caseId,'calibrated-v5'),v6=checkAuthoringBenchmark(fixture,caseId,'calibrated-v6');
        const differences=v5.checks.flatMap((before,i)=>JSON.stringify(before)===JSON.stringify(v6.checks[i])?[]:[{id:before.id,before,after:v6.checks[i]}]);

      } finally {if(fs.realpathSync(fixture)!==fixture||!fixture.startsWith(scratch+'/fixture-'))throw Error('UNSAFE_FIXTURE_CLEANUP');fs.rmSync(fixture,{recursive:true});removed++}
    }
  }
} finally {if(fs.realpathSync(scratch)!==scratch)throw Error('UNSAFE_SCRATCH_CLEANUP');fs.rmSync(scratch,{recursive:true})}

const oraclePath=join(repo,'ai/flow-eval/authoring-benchmark.ts');
const report={schema:'gennady.authoring-pending-calibration.1',primaryGradesReplaced:false,semanticGradesReplaced:false,modelCalls:0,oraclePath,oracleSha256:sha(bytes(oraclePath)),scriptSha256:sha(bytes(fileURLToPath(import.meta.url))),sourceNote:'Post-series v6 alias/terminator calibration; both frozen live arms remain v5. console.log(JSON.stringify({records:rows.length,changed:rows.filter(r=>r.differences.length).map(r=>({index:r.index,mode:r.mode,checks:r.differences.map(d=>d.id)})),cleanup:report.cleanup,modelCalls:0}));
