function dist(root: string) {
  const result: object[] = [];
  function walk(dir: string) {
    for (const name of fs.readdirSync(dir).sort()) {
      const path = join(dir,name), stat = fs.lstatSync(path);
      if (stat.isSymbolicLink()) fail('DIST_ALIAS');
      if (stat.isDirectory()) walk(path);
      else result.push({ path: relative(join(root,'dist'),path), bytes: stat.size, sha256: hash(bytes(path)) });
    }
  }
  if (!fs.existsSync(join(root,'dist/gennady.js'))) fail('NO_BUILD');
  walk(join(root,'dist'));
  return { sha256: hash(JSON.stringify(result)), totalBytes: result.reduce((n,item:any)=>n+item.bytes,0), files: result };
}
