// A separate derived observation, never a mutation of the frozen grade or comparator.
function contractObservations(summary: any, trajectory: any, cleanup: any, caseId: string) {
  const metrics = summary.authoringBenchmark?.metrics;
  const count = (key: string) => Number.isSafeInteger(metrics?.[key]) && metrics[key] >= 0 ? metrics[key] : null;
  const wholeWrites = count('wholeWrites'), partialWrites = count('partialWrites'), outsideTargetWrites = count('outsideTargetWrites');
  const flags = { multipleCompletedNativeTargetWrites: wholeWrites === null ? null : wholeWrites > 1, partialWriteToolAttempts: partialWrites === null ? null : partialWrites > 0, outsideTargetWriteAttempts: outsideTargetWrites === null ? null : outsideTargetWrites > 0 };
  const workerSession = trajectory.events?.find((event: any) => event.type === 'tool' && event.sessionId)?.sessionId;
  const cwd = cleanup.sessions?.find((session: any) => session.id === workerSession)?.directory;
  const target = cwd ? resolve(cwd, getAuthoringCase(caseId).target) : undefined;
  const callEvidence = (trajectory.events ?? []).flatMap((event: any) => {
    if (event.type !== 'tool') return [];
    let path: string | undefined;
    try { const args = JSON.parse(event.arg ?? '{}'); if (cwd && typeof args.filePath === 'string') path = resolve(cwd, args.filePath); } catch { /* Unknown paths stay unknown. */ }
    const related = [];
    if (target && event.tool === 'write' && event.status === 'completed' && path === target) related.push('completed-native-target-write');
    if (['edit', 'apply_patch', 'patch'].includes(event.tool)) related.push('partial-write-tool-attempt');
    if (target && path && ['write', 'edit'].includes(event.tool) && path !== target) related.push('outside-target-write-attempt');
    return related.length ? [{ callId: event.callId ?? null, tool: event.tool, status: event.status ?? 'unknown', flags: related }] : [];
  });
  return { authority: 'derived observation of frozen metrics, not a new primary grade', counts: { wholeWrites, partialWrites, outsideTargetWrites }, flags, observedViolation: Object.values(flags).some(value => value === true), metricFieldsKnown: Object.values(flags).filter(value => value !== null).length, traceCaptureComplete: trajectory.coverage?.complete === true, targetPathBound: Boolean(target), callEvidence, absenceInference: 'No absent/zero metric proves a clean path; opaque Bash, incomplete capture and unknown identity remain unknown. An observed native violation is not erased by an unrelated unknown path verdict. Partial/outside counts are attempts, not proof every attempted mutation succeeded.' };
}
// Missing/inconclusive semantic verdicts and unproven paths remain unknown, not failed grades.
function knownQuality(entries: any[]) {
  const classify = (kind: string, row: any) => {
    if (row.status !== 'completed') return 'unknown';
    if (kind === 'artifact') return row.artifact?.pass === true ? 'pass' : row.artifact?.pass === false ? 'fail' : 'unknown';
    if (kind === 'semantic') return ['pass', 'fail'].includes(row.verdict) ? row.verdict : 'unknown';
    if (row.path?.pass === true) return 'pass';
    // A false path flag is not a typed failure: opaque/unproven paths remain unknown.
    // Positive write violations are a separate selection observation, never a new path grade.
    return 'unknown';
  };
  return Object.fromEntries(['artifact', 'semantic', 'path'].map(kind => {
    const states = entries.map(row => classify(kind, row));
    const passed = states.filter(s => s === 'pass').length, failed = states.filter(s => s === 'fail').length;
    return [kind, { passed, failed, known: passed + failed, unknown: 6 - passed - failed, recorded: entries.length, expected: 6 }];
  }));
}

function describePairCase(left: any, right: any, comparison: any, contract: any, leftKnown: any, rightKnown: any, globalComparison: any = comparison) {
  const dimensions = ['tools', 'workerNonCacheTokens', 'workerMessages', 'workerMs'];
  const qualified = comparison?.comparable === true && left?.recorded === 6 && right?.recorded === 6 && left?.cleanlyBound === 6 && right?.cleanlyBound === 6 && dimensions.every(key => left?.[key]?.complete === true && right?.[key]?.complete === true);
  const delta = (before: any, after: any) => typeof before === 'number' && Number.isFinite(before) && typeof after === 'number' && Number.isFinite(after) ? { baseline: before, candidate: after, absolute: after - before, percent: before > 0 ? (after - before) * 100 / before : null } : null;
  const metricDeltas = Object.fromEntries(dimensions.map(key => [key, delta(left?.[key]?.median, right?.[key]?.median)]));
  const standard = comparison?.cases?.find((c: any) => c.caseId === right?.caseId);
  const promptBytes = delta(standard?.baseline?.promptBytes, standard?.candidate?.promptBytes);
  const quality = Object.fromEntries([['artifactPass', 'artifact'], ['semanticPass', 'semantic'], ['pathPass', 'path']].map(([key, kind]) => [key, { baseline: leftKnown?.[kind!]?.passed ?? null, candidate: rightKnown?.[kind!]?.passed ?? null, baselineFailed: leftKnown?.[kind!]?.failed ?? null, candidateFailed: rightKnown?.[kind!]?.failed ?? null, baselineKnown: leftKnown?.[kind!]?.known ?? 0, candidateKnown: rightKnown?.[kind!]?.known ?? 0, baselineUnknown: leftKnown?.[kind!]?.unknown ?? 6, candidateUnknown: rightKnown?.[kind!]?.unknown ?? 6, expected: 6 }]));
  const qualityComplete = Object.values(quality).every((q: any) => q.baselineKnown === 6 && q.candidateKnown === 6);
  const qualityRegressions = Object.entries(quality).filter(([, q]: any) => q.baselineKnown === 6 && q.candidateFailed > q.baselineFailed).map(([key]) => key);
  const qualityImprovements = Object.entries(quality).filter(([, q]: any) => q.candidateKnown === 6 && q.baselineFailed > q.candidateFailed).map(([key]) => key);
  const direction = (keys: string[]) => {
    if (!qualified || keys.some(key => !metricDeltas[key])) return 'insufficient-data';
    const down = keys.some(key => metricDeltas[key].absolute < 0), up = keys.some(key => metricDeltas[key].absolute > 0);
    return down && up ? 'mixed' : down ? 'better' : up ? 'worse' : 'unchanged';
  };
  const better: string[] = [], worse: string[] = [];
  if (qualified) {
    for (const key of dimensions) { const change = metricDeltas[key]?.absolute; if (change < 0) better.push(key); if (change > 0) worse.push(key); }
    better.push(...qualityImprovements);
    worse.push(...qualityRegressions);
  }
  const effect = !qualified ? 'insufficient-data' : better.length && worse.length ? 'mixed' : better.length ? 'better' : worse.length ? 'worse' : 'mixed';
  const reasons: string[] = [];
  const knownWriteViolation = contract?.adoptionBlockedByObservedContract === true;
  if (!qualified) reasons.push('Incomplete/unqualified six-repetition comparison: displayed local metrics are observations, not a qualified arm effect.');
  if (knownWriteViolation) reasons.push('Observed frozen campaign write criterion violation blocks selection; this is not attribution to instruction format or a global ban on correction rewrites.');
  if (comparison.regressions?.includes(right?.caseId)) reasons.push('Existing standard comparator flags this case; its unchanged pass-count comparison may include unknown semantic/path outcomes and is not alone proof of a quality failure.');
  const noQualityLoss = qualified && qualityComplete ? qualityRegressions.length === 0 : null;
  const noToolMessageGrowth = qualified && metricDeltas.tools?.absolute <= 0 && metricDeltas.workerMessages?.absolute <= 0;
  const tokenTarget = qualified && metricDeltas.workerNonCacheTokens?.percent !== null && metricDeltas.workerNonCacheTokens.percent <= -10;
  const pathProven = rightKnown?.path?.passed === 6;
  if (qualityRegressions.length) reasons.push(`Observed known failures increased with a fully known reference: ${qualityRegressions.join(', ')}. Inconclusive/opaque unknown outcomes were not counted as failure.`);
  if (!qualityComplete) reasons.push('Semantic/artifact/path evidence contains unknown outcomes; neither a drop in confirmation count nor zero PASS counts establish a quality regression, improvement or parity.');
  if (qualified && !noToolMessageGrowth) reasons.push('Tool or assistant-message median grew.');
  if (qualified && !tokenTarget) reasons.push('Median worker non-cache token reduction did not reach 10%.');
  if (!pathProven) reasons.push('Candidate path is not proven 6/6; unknown is not converted to failure or PASS.');
  if (qualified && globalComparison.recommendation !== 'review-measured-results') reasons.push('Existing global comparator/static-envelope selection criterion is not met; local resource metrics remain separately visible.');
  const knownStandardResourceRegression = qualified && standard && (standard.candidate.totalTools > standard.baseline.totalTools || standard.candidate.maxTools > standard.baseline.maxTools || standard.candidate.medianTools > standard.baseline.medianTools || standard.candidate.promptBytes > standard.baseline.promptBytes);
  const knownRegression = qualified && (qualityRegressions.length > 0 || !noToolMessageGrowth || knownStandardResourceRegression);
  const keep = qualified && !knownWriteViolation && !knownRegression && noQualityLoss === true && noToolMessageGrowth && tokenTarget && pathProven && globalComparison.comparable === true && globalComparison.recommendation === 'review-measured-results';
  const action = knownWriteViolation || knownRegression ? 'reject' : keep ? 'keep' : 'keep-original';
  return { caseId: right?.caseId, qualified, globalComparable: globalComparison.comparable === true, effect, effectMeaning: qualified && !better.length && !worse.length ? 'No observed directional difference; grouped as mixed/no-gain, not improvement.' : 'Descriptive resource contrast plus known-quality changes only; unknown quality is not parity, and no causal/statistical guarantee is made.', qualityAssessment: qualityRegressions.length ? 'known-failure-regression' : qualityImprovements.length ? 'known-failure-reduction' : qualityComplete ? 'known-parity' : 'unknown', workerTokenDirection: direction(['workerNonCacheTokens']), trajectoryDirection: direction(['tools', 'workerMessages']), elapsedDirection: direction(['workerMs']), metricDeltas, promptBytes, quality, better, worse, knownWriteViolation, candidateObservedViolationAttempts: contract?.observedViolationAttempts ?? null, candidatePathProven: pathProven, acceptance: { qualityEvidenceComplete: qualityComplete, noQualityLoss, noToolMessageGrowth, tokenTargetAtLeastTenPercent: tokenTarget, standardComparatorRecommendation: globalComparison?.recommendation ?? 'unknown' }, action, actionMeaning: action === 'keep' ? 'Candidate for further review only, not ship/adopt/merge.' : action === 'reject' ? 'Do not select this treatment under the frozen campaign acceptance.' : 'No experimental prompt selected; not a claim that the baseline is perfect.', reasons };
}

}
