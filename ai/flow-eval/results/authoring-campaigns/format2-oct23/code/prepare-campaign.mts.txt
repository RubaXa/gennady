function bytes(path: string, max = 8 * 1024 * 1024): Buffer {
  const st = fs.lstatSync(path);
  if (!st.isFile() || st.isSymbolicLink() || st.size > max) fail('UNSAFE_INPUT_FILE');
  return fs.readFileSync(path);
}

function actionSpan(text: string) {
  const marker = '<Step id="STEP_2_FILL">';
  const first = text.indexOf(marker);
  if (first < 0 || text.indexOf(marker, first + marker.length) >= 0) fail('AMBIGUOUS_STEP_2');
  const end = text.indexOf('</Step>', first);
  const startAction = text.indexOf('<Action>', first) + '<Action>'.length;
  const endAction = text.indexOf('</Action>', startAction);
  if (startAction < first || endAction < startAction || endAction > end) fail('INVALID_ACTION_BOUNDARY');
  return { action: text.slice(startAction, endAction), prefix: text.slice(0, startAction), suffix: text.slice(endAction) };
}

function distManifest(root: string) {
  const dist = join(root, 'dist');
  if (!fs.existsSync(join(dist, 'gennady.js'))) fail('SOURCE_BUILD_MISSING');
  const files: Array<{ path: string; bytes: number; sha256: string }> = [];
  let total = 0;
  const walk = (directory: string) => {
    for (const name of fs.readdirSync(directory).sort()) {
      const path = join(directory, name);
      const st = fs.lstatSync(path);
      if (st.isSymbolicLink()) fail('DIST_SYMLINK');
      if (st.isDirectory()) walk(path);
      else if (st.isFile()) {
        const data = bytes(path, 16 * 1024 * 1024);
        total += data.length;
        if (files.length >= 2048 || total > 64 * 1024 * 1024) fail('DIST_BOUND_EXCEEDED');
        files.push({ path: relative(dist, path).split('\\').join('/'), bytes: data.length, sha256: sha(data) });
      } else fail('DIST_NOT_REGULAR');
    }
  };
  walk(dist);
  return { sha256: sha(JSON.stringify(files)), totalBytes: total, files };
}

// Seeded Fisher–Yates base permutation per case, then cyclic rotations across repetitions.
// Partial position counterbalance: 3 distinct positions/arm, not a full 4-repetition Latin square.
// No Math.random, timing, model result or provider value contributes to the published order.
function orderFor(caseId: string, repetition: number) {
  const armOrder: Array<typeof ARMS[number]> = [...ARMS];
  let counter = 0;
  const bounded = (bound: number) => {
    const limit = Math.floor(2 ** 32 / bound) * bound;
    while (true) {
      const word = createHash('sha256').update(`${SEED}|${caseId}|base|${counter++}`).digest().readUInt32BE(0);
      if (word < limit) return word % bound;
    }
  };
  for (let i = armOrder.length - 1; i > 0; i--) {
    const j = bounded(i + 1);
    [armOrder[i], armOrder[j]] = [armOrder[j]!, armOrder[i]!];
  }
  const offset = (repetition - 1) % armOrder.length;
  return [...armOrder.slice(offset), ...armOrder.slice(0, offset)];
}
