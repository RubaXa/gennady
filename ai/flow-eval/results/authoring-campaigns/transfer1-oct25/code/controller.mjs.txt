function distManifest(root) {
  const dist = join(root, 'dist'), files = [];
  let totalBytes = 0;
  if (lstatSync(dist).isSymbolicLink() || realpathSync(dist) !== dist) fail('DIST_ROOT_ALIAS');
  function walk(directory, depth = 0) {
    if (depth > 32) fail('DIST_DEPTH_EXCEEDED');
    for (const name of readdirSync(directory).sort()) {
      const path = join(directory, name), s = lstatSync(path);
      if (s.isSymbolicLink()) fail('DIST_SYMLINK');
      if (s.isDirectory()) walk(path, depth + 1);
      else if (s.isFile()) {
        if (s.size > 16 * 1024 * 1024 || files.length >= 2048 || totalBytes + s.size > 64 * 1024 * 1024) fail('DIST_BOUND_EXCEEDED');
        const data = readFileSync(path);
        totalBytes += data.length;
        files.push({ path: relative(dist, path).split('\\').join('/'), bytes: data.length, sha256: hash(data) });
      } else fail('DIST_NOT_REGULAR');
    }
  }
  walk(dist);
  return { sha256: hash(JSON.stringify(files)), totalBytes, files };
}

function publicProxyRouting(configuration) {
  // The live server's provider configuration is authoritative, never stale proxy env values.
  // Return public route/key names only: apiKey and header values must not enter evidence.
  const provider = configuration.provider?.['llm-proxy'];

  try { endpoint = new URL(options.baseURL); } catch { fail('AI_PROXY_BASE_URL_INVALID'); }
  const pathname = endpoint.pathname.endsWith('/') ? endpoint.pathname.slice(0, -1) : endpoint.pathname;
  if (endpoint.protocol !== 'https:' || endpoint.origin !== 'https://ai-proxy.vk.team' || pathname !== '/v1' || endpoint.username || endpoint.password || endpoint.search || endpoint.hash) fail('AI_PROXY_ROUTE_NOT_APPROVED');

}
function qualifyPrimary(summary, trajectory, judgeDeadlineObserved = false) {
  const stopReasons = [], warnings = [];
  const coverage = trajectory.coverage;
  if (coverage?.complete !== true || !Number.isInteger(coverage.sessions) || coverage.sessions < 1 || !Array.isArray(coverage.reasons) || coverage.reasons.length || !Array.isArray(trajectory.events)) stopReasons.push('TRACE_CAPTURE_INCOMPLETE');
  if (summary.authoringBenchmark?.environmentVerified !== true) stopReasons.push('ENVIRONMENT_IDENTITY_UNVERIFIED');
  const usage = summary.usage;
  if (!Array.isArray(usage?.models) || usage.models.length !== 1 || usage.models[0] !== MODEL) stopReasons.push('ACTUAL_WORKER_MODEL_UNVERIFIED_OR_WRONG');
  if (usage?.sessions !== coverage?.sessions) stopReasons.push('USAGE_TRACE_SESSION_MISMATCH');
  const terminalFailure = summary.outcome === 'budget-exhausted' || summary.verdict === 'worker-error' || summary.status === 'error';
  const metricsQualified = summary.authoringBenchmark?.modelVerified === true && usage?.complete === true;
  if (!metricsQualified) {
    if (terminalFailure) warnings.push('TERMINAL_FAILURE_PARTIAL_USAGE_NOT_ELIGIBLE_FOR_COST_OR_PASS_CLAIMS');
    else stopReasons.push('COMPLETED_ATTEMPT_MODEL_OR_USAGE_UNVERIFIED');
  }
  if (summary.hasJudge) {
    const judge = summary.judgeUsage?.usage;
    const models = judge?.models;
    const actualWrongModel = Array.isArray(models) && models.length > 0 && (models.length !== 1 || models[0] !== MODEL);
    const actualModelVerified = Array.isArray(models) && models.length === 1 && models[0] === MODEL;
    const qualifiedDeadline = judgeDeadlineObserved && summary.verdict === 'inconclusive';
    if (actualWrongModel || !actualModelVerified && !qualifiedDeadline) stopReasons.push('ACTUAL_JUDGE_MODEL_UNVERIFIED_OR_WRONG');
    else if (!actualModelVerified) warnings.push('JUDGE_DEADLINE_MODEL_USAGE_UNKNOWN_NOT_ZERO');
    if (judge?.complete !== true) warnings.push('JUDGE_USAGE_INCOMPLETE');
  }
  return { dispatchMayContinue: stopReasons.length === 0, metricsQualified, judgeDeadlineObserved, stopReasons, warnings };
}

        rmSync(directory, { recursive: true }); record.tempRemoved = true;
      } catch (error) { record.cleanupErrors.push(safeCode(error)); }
    }
    record.endedAt = new Date().toISOString(); save();
    SddEvalOpenCodeRuntime.prototype.createSession = originalCreate;
    SddEvalOpenCodeRuntime.prototype.judge = originalJudge;
    process.removeListener('SIGINT', onInt); process.removeListener('SIGTERM', onTerm);
    process.exitCode = record.interrupted ? record.interrupted === 'SIGINT' ? 130 : 143 : !record.persistence || !record.tempRemoved || record.cleanupErrors.length || thrown || record.qualification?.dispatchMayContinue !== true ? 2 : 0;
    console.log(JSON.stringify({ controllerAttempt: index, scenarioId: record.scenarioId, persistence: record.persistence, cleanup: record.tempRemoved, cleanupErrors: record.cleanupErrors.length, ...record.primary }));
  }
} else {
  if (existsSync(protocolFile) || existsSync(join(ownRoot, 'records')) || existsSync(join(ownRoot, 'logs'))) fail('CAMPAIGN_ALREADY_STARTED');
  if (existsSync(permanent) && readdirSync(permanent).length) fail('PERMANENT_RESULTS_NOT_EMPTY');
  mkdirSync(permanent, { recursive: true }); mkdirSync(join(ownRoot, 'records')); mkdirSync(join(ownRoot, 'logs')); mkdirSync(join(ownRoot, 'runs'));
  writeFileSync(protocolFile, JSON.stringify(protocol, null, 2) + '\n', { flag: 'wx', mode: 0o600 });
  const { safeVerifyText } = await import(pathToFileURL(join(importSource, 'shared/verify/reporting/report-safety.ts')).href);
  let active, interrupted = null, errors = 0, dispatched = 0;
  const results = [];
  const stop = signal => { interrupted ??= signal; active?.kill(signal); };
  const onInt = () => stop('SIGINT'), onTerm = () => stop('SIGTERM');
  process.on('SIGINT', onInt); process.on('SIGTERM', onTerm);
  try {
    for (let index = 0; index < config.attempts.length && !interrupted; index++) {
      const attempt = config.attempts[index];
      verifySource(attempt.sourceRoot);
      let capture = '', truncated = false;
      const child = spawn(process.execPath, ['--import', join(attempt.sourceRoot, 'node_modules/tsx/dist/loader.mjs'), ownFile, '--config', configPath, '--case', String(index)], { cwd: attempt.sourceRoot, env: { ...gitEnvironment, GENNADY_AUTHORING_TRANSFER_CASE: attempt.taskId }, stdio: ['ignore', 'pipe', 'pipe'] });
      active = child; dispatched++;
      const collect = chunk => { const part = String(chunk); if (Buffer.byteLength(capture) + Buffer.byteLength(part) > LOG_BYTES) { truncated = true; return; } if (!truncated) capture += part; };
      child.stdout.on('data', collect); child.stderr.on('data', collect);
      let hardTimer;
      const timer = setTimeout(() => { child.kill('SIGTERM'); hardTimer = setTimeout(() => child.kill('SIGKILL'), 60000); }, 540000);
      const outcome = await new Promise(resolveChild => { child.once('error', () => resolveChild({ code: null, signal: 'SPAWN_ERROR' })); child.once('close', (code, signal) => resolveChild({ code, signal })); });
      clearTimeout(timer); clearTimeout(hardTimer); active = undefined;
      if (truncated) capture = capture.slice(0, Math.max(0, capture.lastIndexOf('\n'))) + '\n[truncated controller log]\n';
      writeFileSync(join(ownRoot, 'logs', `${index}.log`), safeVerifyText(capture, attempt.sourceRoot), { flag: 'wx', mode: 0o600 });
      const recordPath = join(ownRoot, 'records', `${index}.json`);
      const record = existsSync(recordPath) ? boundedJson(recordPath) : null;
      const complete = record?.persistence && record.tempRemoved && !record.cleanupErrors.length && record.qualification?.dispatchMayContinue === true;
      const row = { index, scenarioId: attempt.scenario.id, ...outcome, complete: Boolean(complete), primary: record?.primary || null, qualification: record?.qualification || null };
      results.push(row);
      atomic(join(ownRoot, 'batch.json'), { planned: config.attempts.length, dispatched, recorded: results.length, interrupted, complete: results.length === config.attempts.length && results.every(r => r.complete), rows: results });
      console.log(JSON.stringify(row));
      if (!complete || outcome.code !== 0) { errors++; break; }
    }
  } finally {
    process.removeListener('SIGINT', onInt); process.removeListener('SIGTERM', onTerm);
    atomic(join(ownRoot, 'batch.json'), { planned: config.attempts.length, dispatched, recorded: results.length, interrupted, errors, complete: results.length === config.attempts.length && results.every(r => r.complete), rows: results, endedAt: new Date().toISOString() });
  }
  process.exitCode = interrupted ? interrupted === 'SIGINT' ? 130 : 143 : results.length !== config.attempts.length || results.some(r => !r.complete) ? 2 : 0;
}
