// @file: Finite EA-3 controller: exact runtimes, no-model qualification, existing eval CLI, owned cleanup.
// @spec: AI-SKILLS
// @consumers: execution-autonomy campaign operator; no automatic repetitions or model fallback

import { execFileSync, spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { existsSync, mkdirSync, mkdtempSync, readFileSync, readdirSync, realpathSync, rmSync, symlinkSync, writeFileSync } from 'node:fs';
import { homedir, tmpdir } from 'node:os';
import { dirname, join, relative, resolve } from 'node:path';
import { setTimeout as delay } from 'node:timers/promises';
import { EXECUTION_AUTONOMY_CASE_IDS, executionAutonomyScenario } from '../execution-autonomy-cases.ts';
import { qualifyExecutionAutonomyCase } from '../execution-autonomy-controls.ts';
import { safeVerifyText } from '../../../shared/verify/reporting/report-safety.ts';

const root = resolve(import.meta.dirname, '../../..');
const mode = process.argv[2];
if (!['--qualify-only', '--live'].includes(mode ?? '')) throw new Error('Require --qualify-only or --live and a new results path');
const out = resolve(root, process.argv[3] ?? 'ai/flow-eval/results/2026-10-26-execution-autonomy');
if (!relative(root, out).startsWith('ai/flow-eval/results/') || existsSync(out)) throw new Error('Require a NEW repository-owned results directory; recorded attempts are never overwritten');
const sha = (bytes: string | Buffer) => createHash('sha256').update(bytes).digest('hex');
const git = (cwd: string, ...args: string[]) => execFileSync('git', args, { cwd, encoding: 'utf8' }).trim();
const json = (path: string, value: unknown) => { mkdirSync(dirname(path), { recursive: true }); writeFileSync(path, JSON.stringify(value, null, 2) + '\n'); };
const owned = realpathSync(mkdtempSync(join(tmpdir(), 'gennady-execution-campaign-')));
const privateDir = join(owned, 'private');
mkdirSync(privateDir, { mode: 0o700 });
mkdirSync(out, { recursive: true });
const records: unknown[] = [];
const controls: unknown[] = [];
let server: ReturnType<typeof spawn> | undefined;
let serverClosed: Promise<void> | undefined;
let failure = false;
let safeCleanup = true;
let stage = 'preregistration';
const stop = new AbortController();
const onSignal = () => stop.abort(new Error('Campaign interrupted; no automatic replacement attempt'));
process.once('SIGINT', onSignal);
process.once('SIGTERM', onSignal);

function filesUnder(path: string): string[] {
  return readdirSync(path, { withFileTypes: true }).flatMap(entry => entry.isDirectory() ? filesUnder(join(path, entry.name)) : entry.isFile() ? [join(path, entry.name)] : []);
}

/** @purpose Freeze the actual distributed runtime rather than assuming dist belongs to HEAD. */
function runtimeIdentity(path: string) {
  const files = ['package.json', 'package-lock.json', ...['dist', 'ai/directives', 'ai/skills'].flatMap(dir => filesUnder(join(path, dir)).map(file => relative(path, file)))].sort().map(file => ({ path: file, sha256: sha(readFileSync(join(path, file))) }));
  return { head: git(path, 'rev-parse', 'HEAD'), tree: git(path, 'rev-parse', 'HEAD^{tree}'), files, digest: sha(JSON.stringify(files)) };
}

/** @purpose Run one existing CLI process without swallowing its lifecycle/partial evidence on a red outcome. */
async function runCli(args: string[], env: NodeJS.ProcessEnv): Promise<number> {
  const child = spawn(process.execPath, ['--import', 'tsx', 'ai/flow-eval/cli.ts', ...args], { cwd: root, env, stdio: 'inherit' });
  const abort = () => child.kill('SIGTERM');
  stop.signal.addEventListener('abort', abort, { once: true });
  try { return await new Promise<number>((done, reject) => { child.once('error', reject); child.once('close', code => done(code ?? 1)); }); }
  finally { stop.signal.removeEventListener('abort', abort); }
}

try {
  const commits = { baseline: git(root, 'rev-parse', '7958ecb0^{commit}'), candidate: git(root, 'rev-parse', 'HEAD') };
  const runtimes = { baseline: join(owned, 'baseline'), candidate: join(owned, 'candidate') };
  const harnessFiles = filesUnder(join(root, 'ai/flow-eval')).filter(path => path.endsWith('.ts') && !path.includes('/results/') && !path.includes('/.results/')).sort();
  const harness = harnessFiles.map(path => ({ path: relative(root, path), sha256: sha(readFileSync(path)) }));
  for (const file of harness) {
    const target = join(out, 'harness', file.path + '.txt');
    mkdirSync(dirname(target), { recursive: true });
    writeFileSync(target, readFileSync(join(root, file.path)));
  }
  const schedule = EXECUTION_AUTONOMY_CASE_IDS.flatMap((caseId, index) => (index % 2 ? ['candidate', 'baseline'] as const : ['baseline', 'candidate'] as const).map(arm => ({ caseId, arm })));
  const protocol = { schema: 'gennady.execution-autonomy-campaign.1', mode, commits, harness, harnessDigest: sha(JSON.stringify(harness)), harnessTrackedDirty: git(root, 'status', '--porcelain') !== '', model: 'llm-proxy/deepseek-v4-flash', planned: mode === '--live' ? 16 : 0, schedule, budgetMs: 480000, repetitions: 1, replacements: 0, concurrency: 1, limitation: 'One sample per case/arm; SDK quiescence is not a sandbox security boundary or proof against arbitrary detached processes. No statistical non-inferiority or universal speed claim. EX4 stops at pending operator decision, not task DONE. Restart and dependency-edge recovery have deterministic controls, not live intervention claims.' };
  json(join(out, 'protocol.json'), protocol);
  for (const arm of ['baseline', 'candidate'] as const) {
    stage = `build:${arm}`;
    stop.signal.throwIfAborted();
    console.log(`[EA-3] build ${arm} ${commits[arm]}`);
    execFileSync('git', ['clone', '--quiet', '--shared', '--no-checkout', root, runtimes[arm]], { stdio: 'pipe' });
    git(runtimes[arm], 'checkout', '--quiet', '--detach', commits[arm]);
    symlinkSync(realpathSync(join(root, 'node_modules')), join(runtimes[arm], 'node_modules'), 'dir');
    const build = execFileSync('npm', ['run', 'build', '--', '--configLoader', 'runner'], { cwd: runtimes[arm], encoding: 'utf8', maxBuffer: 20_000_000 });
    json(join(out, `runtime-${arm}.json`), { ...runtimeIdentity(runtimes[arm]), buildOutputSha256: sha(build), trackedClean: git(runtimes[arm], 'status', '--porcelain', '--untracked-files=no') === '' });
    for (const caseId of EXECUTION_AUTONOMY_CASE_IDS) {
      stage = `qualification:${arm}:${caseId}`;
      stop.signal.throwIfAborted();
      const control = await qualifyExecutionAutonomyCase(runtimes[arm], caseId);
      controls.push({ arm, ...control });
      json(join(out, 'qualification.json'), { modelCalls: 0, controls, complete: controls.length === 16 });
      console.log(`[EA-3] ${arm} ${caseId} normal/broken/tamper/cleanup PASS`);
    }
  }
  if (mode === '--live') {
    stage = 'approved-proxy-and-private-server';
    for (const name of ['LLM_PROXY_BASE_URL', 'LLM_PROXY_API_KEY']) if (!process.env[name]) throw new Error('Approved proxy environment is missing; no model fallback');
    const configured = JSON.parse(readFileSync(join(homedir(), '.config/opencode/opencode.json'), 'utf8'));
    const provider = configured.provider?.['llm-proxy'];
    if (!provider?.options?.apiKey || !provider.models?.['deepseek-v4-flash']) throw new Error('Approved Flash configuration missing');
    const endpoint = new URL(provider.options.baseURL);
    if (endpoint.origin !== 'https://ai-proxy.vk.team' || endpoint.pathname.replace(/\/$/, '') !== '/v1') throw new Error('Proxy route outside operator authorization');
    const config = { model: protocol.model, small_model: protocol.model, enabled_providers: ['llm-proxy'], share: 'disabled', instructions: [], plugin: [], mcp: {}, compaction: { auto: false, prune: false }, permission: { '*': 'allow', external_directory: 'deny', question: 'deny' }, provider: { 'llm-proxy': { ...provider, models: { 'deepseek-v4-flash': provider.models['deepseek-v4-flash'] } } } };
    writeFileSync(join(privateDir, 'opencode.json'), JSON.stringify(config), { mode: 0o600 });
    const net = await import('node:net');
    const reservation = net.createServer();
    await new Promise<void>((done, reject) => reservation.once('error', reject).listen(0, '127.0.0.1', done));
    const port = (reservation.address() as import('node:net').AddressInfo).port;
    await new Promise<void>(done => reservation.close(() => done()));
    if (port < 4097) throw new Error('Unsafe private port');
    const baseUrl = `http://127.0.0.1:${port}`;
    const env: NodeJS.ProcessEnv = { ...process.env, OPENCODE_CONFIG: join(privateDir, 'opencode.json'), OPENCODE_DISABLE_SHARE: 'true' };
    for (const key of ['CONFIG', 'DATA', 'CACHE', 'STATE']) env[`XDG_${key}_HOME`] = join(privateDir, key.toLowerCase());
    server = spawn('opencode', ['serve', '--hostname', '127.0.0.1', '--port', String(port), '--log-level', 'WARN'], { cwd: owned, env, detached: true, stdio: ['ignore', 'pipe', 'pipe'] });
    serverClosed = new Promise(done => server!.once('close', () => done()));
    server.stdout?.resume(); server.stderr?.resume();
    server.once('error', () => stop.abort(new Error('Private OpenCode failed; no model fallback')));
    let ready = false;
    for (let i = 0; i < 60 && !ready; i++) { stop.signal.throwIfAborted(); if (server.exitCode !== null || server.signalCode !== null) throw new Error('Private OpenCode exited before readiness'); ready = await fetch(`${baseUrl}/global/health`, { signal: AbortSignal.timeout(1000) }).then(r => r.ok).catch(() => false); if (!ready) await delay(500); }
    if (!ready) throw new Error('Private server unavailable; no model calls attempted');
    const sandboxRoot = join(owned, 'sandboxes'); mkdirSync(sandboxRoot);
    for (const [index, entry] of schedule.entries()) {
      stage = `live:${index + 1}:${entry.arm}:${entry.caseId}`;
      stop.signal.throwIfAborted();
      if (harness.some(file => sha(readFileSync(join(root, file.path))) !== file.sha256)) throw new Error('Harness drift after preregistration; stop, do not recolor/repeat');
      const scenario = executionAutonomyScenario(entry.caseId, entry.arm);
      const scenarioFile = join(out, 'scenarios', `${index}.json`);
      json(scenarioFile, [scenario]);
      const recordFile = join(out, 'records', `${index}.json`);
      json(recordFile, { ...entry, status: 'started', startedAt: new Date().toISOString() });
      console.log(`[EA-3] live ${index + 1}/16 ${entry.arm} ${entry.caseId}`);
      const exitCode = await runCli(['--scenario-file', scenarioFile, '--directory', sandboxRoot, '--gennady-root', runtimes[entry.arm], '--base-url', baseUrl, '--model', protocol.model, '--judge-model', protocol.model, '--concurrency', '1', '--observe-every-ms', '15000', '--stuck-after', '8', '--max-observations', '40', '--max-wall-clock-ms', '480000', '--results-dir', join(out, 'permanent'), '--artifacts-dir', join(out, 'transient')], env);
      const pending = readdirSync(sandboxRoot).filter(name => name.startsWith('sdd-flow-eval-'));
      const record = { ...entry, exitCode, status: pending.length ? 'cleanup-unproven' : 'recorded', completedAt: new Date().toISOString(), remainingSandboxes: pending };
      json(recordFile, record); records.push(record);
      json(join(out, 'progress.json'), { planned: 16, recorded: records.length, records });
      if (pending.length) { safeCleanup = false; throw new Error('Writer cleanup unproven; campaign stopped, recovery paths retained'); }
      if (!filesUnder(join(out, 'permanent')).some(path => path.endsWith('/summary.json') && path.includes(scenario.id))) throw new Error('Missing primary evidence; no automatic replacement');
    }
  }
} catch (cause) {
  failure = true;
  const detail = safeVerifyText(cause instanceof Error ? cause.message : 'unknown campaign error', root).slice(0, 8000);
  json(join(out, 'failure.json'), { stage, detail, replacements: 0, fallback: false });
  console.error(`[EA-3] stopped at ${stage}: ${detail}; no fallback, replacement or PASS inferred`);
} finally {
  process.removeListener('SIGINT', onSignal); process.removeListener('SIGTERM', onSignal);
  if (server?.pid) {
    try { process.kill(-server.pid, 'SIGTERM'); } catch { /* already exited */ }
    await Promise.race([serverClosed, delay(3000)]);
    try { process.kill(-server.pid, 0); process.kill(-server.pid, 'SIGKILL'); } catch { /* group gone */ }
    await Promise.race([serverClosed, delay(3000)]);
    try { process.kill(-server.pid, 0); safeCleanup = false; } catch { /* owned group is gone */ }
  }
  json(join(out, 'cleanup.json'), { safeCleanup, ownedRoot: safeCleanup ? '<removed>' : owned, modelCallsPlanned: mode === '--live' ? 16 : 0, recorded: records.length });
  if (safeCleanup) rmSync(owned, { recursive: true, force: true });
  else console.error(`[EA-3] recovery retained at ${owned}; do not delete while writers are unproven`);
  json(join(out, 'cleanup.json'), { safeCleanup, ownedRoot: safeCleanup ? '<removed>' : owned, rootRemoved: !existsSync(owned), planned: mode === '--live' ? 16 : 0, recorded: records.length });
}
if (failure) process.exitCode = 1;
