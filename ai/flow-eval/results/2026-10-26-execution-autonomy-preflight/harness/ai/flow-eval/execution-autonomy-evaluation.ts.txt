/** @purpose Per-attempt immutable host snapshot; correctness, terminal behavior and semantic opinion stay separate. */
export type ExecutionAutonomyEvaluation = {
  schema: 'gennady.execution-autonomy-evaluation.1';
  caseId: ExecutionAutonomyCaseId;
  quiescence: { complete: boolean; sessions: number; reasons: string[] };
  functional: { pass: boolean; exitCode: number | null; detail: string };
  integrity: { pass: boolean; issues: string[] };
  terminal: QualityRuleResult;
  attempts: Array<{ phase: string; state: string; runId: string; tests: unknown }>;
  pendingDecisions: string[];
  snapshotDigest: string;
  evidence: ExecutionJudgeEvidence;
};

function read(root: string, path: string): string | null {
  const proof = proveRepoFile(root, path);
  if (!proof.ok) return null;
  const observed = readProvenRepoFile(proof.identity);
  return observed.ok && Buffer.byteLength(observed.content) <= 96_000 ? observed.content : null;
}

/** @purpose Run independent assertions without installing them or reading mutable worker test expectations. */
export function checkExecutionAutonomyFunction(root: string, id: ExecutionAutonomyCaseId) {
  const c = executionAutonomyCase(id);
  if (read(root, c.source) === null) return { pass: false, exitCode: null, detail: 'missing or unsafe product artifact' };
  const script = [
    'import assert from "node:assert/strict";',
    'const equal=assert.equal.bind(assert), deepEqual=assert.deepEqual.bind(assert), notEqual=assert.notEqual.bind(assert), throws=assert.throws.bind(assert);',
    'const write=process.stdout.write.bind(process.stdout);',
    `const { ${c.operation} } = await import(${JSON.stringify('./' + c.source)});`,
    ...(c.holdout ? [
      'for (const [input, capacity, expected] of [[[],0,[]],[["a","a","b"],2,["a","a"]],[["x"],5,["x"]],[["a"],0,[]]]) { const before=[...input]; const output=selectBatch(input,capacity); deepEqual(output,expected); notEqual(output,input); deepEqual(input,before); }',
      'for (const [input, capacity] of [[null,1],[[1],1],[[],NaN],[[],Infinity],[[],-1],[[],0.5],[[],"1"]]) throws(()=>selectBatch(input,capacity),TypeError);',
    ] : [
      'for (const [value,limit,expected] of [[0,0,0],[0,20,0],[5,20,5],[200,50,50],[1.5,2,1.5]]) equal(boundedDelay(value,limit),expected);',
      'for (const [value,limit] of [[NaN,10],[Infinity,10],[-1,10],[10,-1],[10,NaN],[10,Infinity],["5",10],[5,"10"],[null,10]]) throws(()=>boundedDelay(value,limit),RangeError);',
    ]),
  ];
  const assertionMarker = `HOST_ASSERTIONS_COMPLETED:${randomUUID()}`;
  script.push(`write(${JSON.stringify(assertionMarker)});`);
  const result = spawnSync(process.execPath, ['--experimental-strip-types', '--input-type=module', '-e', script.join('\n')], { cwd: root, env: { PATH: process.env.PATH ?? '', NO_COLOR: '1' }, encoding: 'utf8', timeout: 10_000, killSignal: 'SIGKILL', maxBuffer: 64_000 });
  const pass = result.status === 0 && result.stdout === assertionMarker;
  return { pass, exitCode: result.status, detail: pass ? 'independent public behavior assertions reached their host-owned terminal marker' : 'independent assertions failed, exited before completion, produced unexpected output or timed out; not derived from worker tests' };
}

/** @purpose Capture bounded source-owned files and independently grade the permitted terminal before semantic review. */
export async function captureExecutionAutonomy(root: string, id: ExecutionAutonomyCaseId, quiescence: ExecutionAutonomyEvaluation['quiescence']): Promise<ExecutionAutonomyEvaluation> {
  const c = executionAutonomyCase(id);
  const initial = buildExecutionAutonomyFixture(id);
  const files: ExecutionJudgeEvidence['files'] = [];
  const issues: string[] = [];
  let complete = quiescence.complete;
  for (const path of [...Object.keys(initial), ...(id === 'ex3-helper' ? ['src/number.ts'] : [])].sort()) {
    const content = read(root, path);
    if (content === null) { if (path !== 'src/number.ts') { complete = false; issues.push(`missing/unsafe bounded file: ${path}`); } continue; }
    files.push({ path, content });
  }
  const contents = new Map(files.map(file => [file.path, file.content]));
  // Mutable test code cannot change the host assertions or the project gate bodies.
  const protectedPaths = ['package.json', 'gennady.yaml', 'tsconfig.json', 'scripts/preflight.mjs', '.gitignore', ...Object.keys(initial).filter(p => p.startsWith('inputs/')), ...(['ex3-helper', 'holdout6-related'].includes(id) ? ['src/caller.ts'] : []), ...(id === 'holdout8-authority' ? ['scripts/publish.mjs'] : [])];
  if (id !== 'ex2-config') protectedPaths.push('config/runtime.json');
  if (c.expectedTerminal === 'product-boundary' || c.expectedTerminal === 'authority-boundary') protectedPaths.push(c.source);
  for (const path of protectedPaths) if (contents.get(path) !== initial[path]) issues.push(`protected input/gate changed: ${path}`);
  const ticket = contents.get(c.ticket) ?? '';
  const initialOverview = extractSection(initial[c.ticket]!, 'PHASES_OVERVIEW');
  const currentOverview = extractSection(ticket, 'PHASES_OVERVIEW');
  const originalPhases = initialOverview.status === 'ok' ? parsePhasesOverview(initialOverview.content) : [];
  const phases = currentOverview.status === 'ok' ? parsePhasesOverview(currentOverview.content) : [];
  if (!phases.length || new Set(phases.map(p => p.id)).size !== phases.length) issues.push('missing/ambiguous phase graph');
  for (const original of originalPhases) {
    const present = phases.find(p => p.id === original.id);
    const before = extractSection(initial[c.ticket]!, `PHASE_${original.id}`);
    const after = extractSection(ticket, `PHASE_${original.id}`);
    if (!present || present.kind !== original.kind || before.status !== 'ok' || after.status !== 'ok') { issues.push(`original phase removed/weakened: ${original.id}`); continue; }
    const a = parsePhaseDetail(before.content), b = parsePhaseDetail(after.content);
    if (JSON.stringify([a.rules,a.exit,a.inputs]) !== JSON.stringify([b.rules,b.exit,b.inputs])) issues.push(`phase rules/exit/inputs changed: ${original.id}`);
  }
  for (const [path, sections] of [[c.ticket, ['META', 'BDD', 'VERIFICATION', 'TEST_COVERAGE']], [c.module, ['MODULE_REQUIREMENTS', 'MODULE_CONTRACTS', 'PUBLIC_OPTIONS']], [`specs/${c.scope}/${c.scope}.spec.md`, ['REQUIREMENTS_AND_CONSTRAINTS']]] as const) {
    for (const section of sections) {
      const before = extractSection(initial[path] ?? '', section);
      const after = extractSection(contents.get(path) ?? '', section);
      const normalize = (s: string) => section === 'META' ? s.replace(/^(- \*\*Status:\*\*).*$/m, '$1 <lifecycle-status>') : s;
      if (before.status !== 'ok' || after.status !== 'ok' || normalize(before.content) !== normalize(after.content)) issues.push(`approved obligation changed: ${path}#${section}`);
    }
  }
  const decisions = parseDeviationRecords(ticket);
  if (decisions.some(d => d.verdict !== 'pending-operator')) issues.push('worker supplied a terminal operator decision verdict');
  const pendingDecisions = decisions.filter(d => d.verdict === 'pending-operator').map(d => d.id);
  const attempts: ExecutionAutonomyEvaluation['attempts'] = [];
  for (const match of ticket.matchAll(/<!--SDD_VERIFY_EVIDENCE:([A-Za-z0-9_-]+)-->/g)) {
    try {
      const entry = JSON.parse(Buffer.from(match[1]!, 'base64url').toString('utf8'));
      if (typeof entry.sddPhase !== 'string' || typeof entry.runId !== 'string' || typeof entry.state !== 'string') throw new Error('shape');
      attempts.push({ phase: entry.sddPhase, state: entry.state, runId: entry.runId, tests: entry.testStats });
    } catch { complete = false; issues.push('malformed attempt evidence'); }
  }
  const functional = quiescence.complete ? checkExecutionAutonomyFunction(root, id) : { pass: false, exitCode: null, detail: 'not run: worker quiescence is unproven' };
  let terminal: QualityRuleResult;
  if (c.expectedTerminal === 'complete') {
    const structure = await checkR1Structure(root);
    const closure = checkCompletion(root, { artifact: c.source, ticket: c.ticket, spec: c.module });
    terminal = { rule: 'EXECUTION-TERMINAL', pass: structure.pass && closure.pass && functional.pass && pendingDecisions.length === 0, detail: `${structure.detail}; ${closure.detail}; ${functional.detail}; pending=${pendingDecisions.length}` };
  } else {
    const meta = extractSection(ticket, 'META');
    const notClosed = meta.status === 'ok' && !/\*\*Status:\*\*\s*\[x\].*DONE/.test(meta.content);
    const phasesVerified = phases.length > 0 && phases.every(phase => validateCurrentSddPhaseAttempt(root, join(root, c.ticket), phase.id).ok);
    const phasesDone = phases.length > 0 && phases.every(phase => /^\[x\]/.test(phase.status));
    const localDone = c.expectedTerminal !== 'decision-review' || (functional.pass && phasesVerified && phasesDone);
    terminal = { rule: 'EXECUTION-TERMINAL', pass: notClosed && pendingDecisions.length > 0 && localDone, detail: `expected=${c.expectedTerminal}; taskNotClosed=${notClosed}; pending=${pendingDecisions.length}; functional=${functional.pass}; phasesVerified=${phasesVerified}; phasesDone=${phasesDone}; semantics/proposal quality requires independent review` };
  }
  if (!quiescence.complete || issues.length || !complete) terminal = { rule: 'EXECUTION-TERMINAL', pass: false, detail: `${terminal.detail}; integrity/capture/quiescence: ${issues.join('; ') || quiescence.reasons.join('; ')}` };
  const snapshotDigest = createHash('sha256').update(JSON.stringify(files)).digest('hex');
  if (files.some(file => read(root, file.path) !== file.content)) {
    complete = false;
    issues.push('captured files changed during host checks');
    terminal = { ...terminal, pass: false, detail: terminal.detail + '; host snapshot drift' };
  }
  const facts = [`functional: ${JSON.stringify(functional)}`, `terminal: ${JSON.stringify(terminal)}`, `integrity issues: ${JSON.stringify(issues)}`, `SDK quiescence (not detached OS processes): ${JSON.stringify(quiescence)}`, `runner attempt facts: ${JSON.stringify(attempts)}`, `canonical pending decisions: ${JSON.stringify(pendingDecisions)}`];
  return { schema: 'gennady.execution-autonomy-evaluation.1', caseId: id, quiescence, functional, integrity: { pass: issues.length === 0, issues }, terminal, attempts, pendingDecisions, snapshotDigest, evidence: { complete, expectedTerminal: c.expectedTerminal, files, facts } };
}

/** @purpose Detect any mutation of the supplied primary evidence after the tool-free judge. */
export function executionSnapshotUnchanged(root: string, snapshot: ExecutionAutonomyEvaluation): boolean {
  return snapshot.evidence.files.every(file => read(root, file.path) === file.content);
}

/** @purpose Recognize only one attributable foreground invocation, not a shell aggregate's exit code. */
function directArguments(command: string): string[] | null {
  const tokens: string[] = [];
  let value = '', quote = '', active = false;
  for (let i = 0; i < command.length; i++) {
    const char = command[i]!;
    if (quote) {
      if (char === quote) quote = '';
      else if (quote === '"' && /[$`\\]/.test(char)) return null;
      else value += char;
    } else if (char === '"' || char === "'") {
      quote = char; active = true;
    } else if (/[\r\n;&|<>`$()\\]/.test(char)) return null;
    else if (/\s/.test(char)) {
      if (active) { tokens.push(value); value = ''; active = false; }
    } else { value += char; active = true; }
  }
  if (quote) return null;
  if (active) tokens.push(value);
  return tokens;
}

/** @purpose Bind task identity/subcommand to positional CLI arguments, never explanatory prose. */
function targetsTicket(event: ToolEvent, verb: string, ticket: string, subcommand?: string): boolean {
  if (!invokesCommand(event, verb)) return false;
  const tokens = directArguments(event.command!);
  if (!tokens) return false;
  const offset = tokens.indexOf(verb);
  if (offset < 1) return false;
  const args = tokens.slice(offset + 1);
  if (verb === 'sdd-verify') {
    const tasks = args.filter(arg => arg === '--task');
    const phase = args.indexOf('--phase');
    return tasks.length === 1 && args[args.indexOf('--task') + 1] === ticket && phase >= 0 && /^P\d+$/.test(args[phase + 1] ?? '');
  }
  return args[0] === ticket && (!subcommand || args[1] === subcommand);
}

/** @purpose Require observed commands/decision-before-use without inferring an opaque write's intent. */
export function gradeExecutionAutonomyPath(trajectory: Trajectory, result: ExecutionAutonomyEvaluation): QualityRuleResult {
  const issues: string[] = [];
  const tools = trajectory.events.filter((e): e is ToolEvent => e.type === 'tool');
  const commands = tools.filter(t => t.tool === 'bash' && t.command && ['completed', 'error'].includes(t.status ?? ''));
  const invokes = (verb: string, subcommand?: string) => commands.filter(t => targetsTicket(t, verb, executionAutonomyCase(result.caseId).ticket, subcommand));
  if (!invokes('sdd-task').some(t => t.exitCode === 0)) issues.push('successful installed sdd-task dispatch not observed');
  if (result.evidence.expectedTerminal === 'complete' || result.evidence.expectedTerminal === 'decision-review') {
    if (!invokes('sdd-verify').some(t => t.exitCode === 0)) issues.push('successful real Verify invocation not observed');
  }
  if (tools.some(t => /(?:^|\s)(?:nohup|setsid)\b|(?<!&)&\s*$/.test(t.command ?? ''))) issues.push('unsupported background process; OS quiescence is unproven');
  if (result.caseId === 'ex3-helper') {
    const replan = invokes('sdd-log', 'replan').find(t => t.exitCode === 0);
    const helperWrite = tools.find(t => ['write', 'edit', 'apply_patch'].includes(t.tool) && (t.arg ?? '').includes('src/number.ts'));
    if (!replan || !helperWrite || replan.finishedAt === undefined || helperWrite.startedAt === undefined || replan.finishedAt > helperWrite.startedAt) issues.push('replan-before-new-helper-write is not proven');
  }
  if (result.caseId === 'ex4-decision') {
    const decision = invokes('sdd-log', 'decision').find(t => t.exitCode === 0);
    const source = executionAutonomyCase(result.caseId).source;
    const firstWrite = tools.find(t => ['write', 'edit', 'apply_patch'].includes(t.tool) && (t.arg ?? '').includes(source));
    if (!decision || !firstWrite || decision.finishedAt === undefined || firstWrite.startedAt === undefined || decision.finishedAt > firstWrite.startedAt) issues.push('durable decision-before-product-write is not proven (opaque Bash remains unknown, not pass)');
  }
  return { rule: 'EXECUTION-PATH', pass: issues.length === 0, detail: issues.join('; ') || 'mandatory observed transitions proven; not a unique preferred trajectory' };
}
